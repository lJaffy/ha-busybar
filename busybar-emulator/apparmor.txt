#include <tunables/global>

profile busybar_emulator flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  file,
  signal (send) set=(kill,term,int,hup,cont),

  # S6 overlay (s6-rc layout; user bundle lives in s6-rc.d, which S6 writes at startup)
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Options and other files of the app
  /data/** rw,

  # Run the emulator stack in its own, stricter profiles. Transition targets
  # must be exact paths: a glob here fails the compile with "merged rule
  # with conflicting x modifiers".
  /usr/bin/node cx -> emulator_node,
  /usr/bin/python3 cx -> emulator_relay,
  /usr/sbin/nginx cx -> emulator_nginx,

  profile emulator_node flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Receive signals from S6 overlay
    signal (receive) peer=*_busybar_emulator,

    # Emulator code and its persistent data
    /opt/busy-emulator/** r,
    /data/** rw,
    /tmp/** rwk,

    # Node runtime needs exec of itself + proc/self maps
    /usr/bin/node rmix,
    /usr/lib/node_modules/** r,
    /proc/@{pid}/** r,
    /sys/devices/system/cpu/online r,
    /dev/urandom r,
    /dev/null rw,
    /dev/tty rw,

    # Outbound TCP (LAN clients connect in; emulator fetches app URLs)
    network inet tcp,
    network inet6 tcp,
  }

  profile emulator_relay flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    signal (receive) peer=*_busybar_emulator,

    /usr/bin/busy-relay r,
    /usr/bin/python3 rmix,
    /usr/lib/python3*/** r,
    /data/** rw,
    /dev/null rw,
    /dev/tty rw,

    network inet tcp,
  }

  profile emulator_nginx flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    signal (receive) peer=*_busybar_emulator,

    # nginx runs as root here (no user switching), so it needs passwd/group
    # lookups and ownership-neutral paths; logs go to stderr, pid to /run.
    capability net_bind_service,
    /usr/sbin/nginx rmix,
    /etc/nginx/** r,
    /run/nginx.pid rw,
    /run/nginx/** rwk,
    /dev/null rw,
    /dev/tty rw,
    /dev/stderr w,
    /dev/stdout w,

    network inet tcp,
  }
}
